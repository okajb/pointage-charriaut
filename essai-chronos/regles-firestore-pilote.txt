rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function isEmployee() {
      return request.auth != null
        && request.auth.token.email_verified == true
        && request.auth.token.email is string
        && exists(/databases/$(database)/documents/authorizedEmployees/$(request.auth.token.email))
        && get(/databases/$(database)/documents/authorizedEmployees/$(request.auth.token.email)).data.active == true;
    }

    function isImporter() {
      return isEmployee()
        && get(/databases/$(database)/documents/authorizedEmployees/$(request.auth.token.email)).data.get('role','') == 'admin';
    }

    match /authorizedEmployees/{email} {
      allow get: if isEmployee()
        && request.auth.token.email == email;
      allow list, write: if false;
    }

    match /pumps/{pumpId} {
      allow read: if isEmployee()
        && pumpId in ['testo', 'value'];
      allow write: if false;

      match /history/{recordId} {
        allow read: if isEmployee() && pumpId in ['testo', 'value'];

        allow create: if isImporter()
          && pumpId in ['testo', 'value']
          && recordId.matches('legacy-2026-sheet-r[0-9]{3}-(testo|value)')
          && recordId.matches('.*-' + pumpId)
          && request.resource.data.keys().hasAll([
            'pumpId','type','sourceRow','sourceName','sourceSheet','sourceYear',
            'isHistorical','date','dateAsWritten','durationMinutes','operation',
            'client','gasAsWritten','employee','importedBy','importedAt'
          ])
          && request.resource.data.keys().hasOnly([
            'pumpId','type','sourceRow','sourceName','sourceSheet','sourceYear',
            'isHistorical','date','dateAsWritten','durationMinutes','operation',
            'client','gasAsWritten','employee','importedBy','importedAt'
          ])
          && request.resource.data.pumpId == pumpId
          && request.resource.data.isHistorical == true
          && request.resource.data.sourceYear == 2026
          && request.resource.data.sourceName == 'Temps pompe vide'
          && request.resource.data.sourceSheet == 'Feuille 1'
          && request.resource.data.sourceRow is int
          && request.resource.data.sourceRow >= 2
          && request.resource.data.sourceRow <= 40
          && request.resource.data.type in ['usage','oil']
          && (request.resource.data.date == null ||
              (request.resource.data.date is string &&
               request.resource.data.date.matches('2026-[0-9]{2}-[0-9]{2}')))
          && (request.resource.data.dateAsWritten == null ||
              request.resource.data.dateAsWritten is string)
          && (request.resource.data.operation == null ||
              request.resource.data.operation is string)
          && (request.resource.data.client == null ||
              request.resource.data.client is string)
          && (request.resource.data.gasAsWritten == null ||
              request.resource.data.gasAsWritten is string)
          && (request.resource.data.employee == null ||
              request.resource.data.employee is string)
          && ((request.resource.data.type == 'usage'
               && request.resource.data.durationMinutes is int
               && request.resource.data.durationMinutes > 0
               && request.resource.data.durationMinutes <= 1440)
              ||
              (request.resource.data.type == 'oil'
               && request.resource.data.durationMinutes == null))
          && request.resource.data.importedBy == request.auth.token.email
          && request.resource.data.importedAt == request.time;

        allow update, delete: if false;
      }
    }

    match /pilotPumps/{pumpId} {
      function changedOnly(keys) {
        return request.resource.data.diff(resource.data).affectedKeys().hasOnly(keys);
      }
      function goodInit() {
        return isImporter()
          && resource.data.get('trackingStartedAt',null) == null
          && resource.data.get('activeSession',null) == null
          && resource.data.get('usedSeconds',0) == 0
          && resource.data.get('lifetimeSeconds',0) == 0
          && request.resource.data.usedSeconds == 0
          && request.resource.data.lifetimeSeconds == 0
          && request.resource.data.trackingStartedAt == request.time
          && request.resource.data.trackingInitializedBy == request.auth.token.email
          && request.resource.data.oilCondition == 'unverified'
          && request.resource.data.lastActionAt == request.time
          && changedOnly(['usedSeconds','lifetimeSeconds','trackingStartedAt',
                          'trackingInitializedBy','oilCondition','lastActionAt']);
      }
      function goodStart() {
        let original = resource.data.get('activeSession', null);
        let next = request.resource.data.get('activeSession',null);
        return original == null
          && resource.data.get('trackingStartedAt',null) is timestamp
          && resource.data.get('usedSeconds',null) is int
          && next is map
          && next.keys().hasOnly(['byUid','byEmail','byName','gas','job','note','startedAt'])
          && next.keys().hasAll(['byUid','byEmail','byName','gas','job','note','startedAt'])
          && next.byUid == request.auth.uid
          && next.byEmail == request.auth.token.email
          && next.byName is string && next.byName.size() > 0 && next.byName.size() <= 90
          && next.gas is string && next.gas.size() > 0 && next.gas.size() <= 80
          && next.job is string && next.job.size() <= 80
          && next.note is string && next.note.size() <= 600
          && next.startedAt == request.time
          && request.resource.data.lastActionAt == request.time
          && changedOnly(['activeSession','lastActionAt']);
      }
      function goodStop() {
        let original = resource.data.get('activeSession',null);
        let next = request.resource.data;
        let evt = getAfter(/databases/$(database)/documents/pilotPumps/$(pumpId)/events/$(next.lastEventId)).data;
        return original is map
          && next.activeSession == null
          && next.lastActionAt == request.time
          && next.lastEventId is string
          && changedOnly(['activeSession','usedSeconds','lifetimeSeconds','lastActionAt','lastEventId'])
          && evt.type == 'usage'
          && evt.at == request.time
          && evt.seconds is int && evt.seconds >= 1 && evt.seconds <= 172800
          && evt.startedAt == original.startedAt
          && evt.byEmail == original.byEmail
          && evt.byUid == original.byUid
          && evt.person == original.byName
          && evt.gas == original.gas
          && evt.job == original.job
          && evt.note == original.note
          && evt.endedByUid == request.auth.uid
          && next.usedSeconds == resource.data.get('usedSeconds',0) + evt.seconds
          && next.lifetimeSeconds == resource.data.get('lifetimeSeconds',0) + evt.seconds
          && request.time >= original.startedAt
          && (request.time - original.startedAt) <= duration.value(evt.seconds + 120,'s')
          && (request.time - original.startedAt) + duration.value(120,'s') >= duration.value(evt.seconds,'s');
      }
      function goodOil() {
        let next = request.resource.data;
        let evt = getAfter(/databases/$(database)/documents/pilotPumps/$(pumpId)/events/$(next.lastEventId)).data;
        return resource.data.get('activeSession',null) == null
          && next.get('activeSession',null) == null
          && next.usedSeconds == 0
          && next.lastOilAt == request.time
          && next.oilCondition == 'changed'
          && next.lastActionAt == request.time
          && next.lastEventId is string
          && changedOnly(['usedSeconds','lastOilAt','oilCondition','lastActionAt','lastEventId'])
          && evt.type == 'oil'
          && evt.at == request.time
          && evt.byEmail == request.auth.token.email
          && evt.byUid == request.auth.uid
          && evt.previousSeconds == resource.data.get('usedSeconds',0);
      }
      allow get: if pumpId in ['testo','value'] && isEmployee();
      allow list, delete: if false;
      allow create: if pumpId in ['testo','value'] && isImporter()
        && request.resource.data.keys().hasAll([
          'name','color','oilIntervalHours','usedSeconds','lifetimeSeconds',
          'activeSession','createdAt','createdBy'
        ])
        && request.resource.data.keys().hasOnly([
          'name','color','oilIntervalHours','usedSeconds','lifetimeSeconds',
          'activeSession','createdAt','createdBy'
        ])
        && request.resource.data.name == (pumpId == 'testo' ? 'TESTO' : 'VALUE')
        && request.resource.data.color == (pumpId == 'testo' ? 'orange' : 'blue')
        && request.resource.data.oilIntervalHours == 10
        && request.resource.data.usedSeconds == 0
        && request.resource.data.lifetimeSeconds == 0
        && request.resource.data.activeSession == null
        && request.resource.data.createdAt == request.time
        && request.resource.data.createdBy == request.auth.token.email;
      allow update: if pumpId in ['testo','value'] && isImporter()
        && (goodInit() || goodStart() || goodStop() || goodOil());
      match /events/{eventId} {
        function linkedToPump() {
          let after = getAfter(/databases/$(database)/documents/pilotPumps/$(pumpId)).data;
          return after.lastEventId == eventId && after.lastActionAt == request.time;
        }
        function validEvent() {
          let e = request.resource.data;
          return e.keys().hasOnly(['type','at','startedAt','person','byEmail','byUid',
                    'endedBy','endedByUid','gas','job','note','seconds','previousSeconds'])
            && e.at == request.time
            && e.person is string && e.person.size() > 0 && e.person.size() <= 90
            && e.byEmail is string && e.byUid is string
            && e.note is string && e.note.size() <= 600
            && ((e.type == 'usage'
                && e.keys().hasAll(['startedAt','seconds','gas','job','endedBy','endedByUid'])
                && e.gas is string && e.job is string
                && e.endedBy is string && e.endedBy.size() <= 90
                && e.endedByUid == request.auth.uid
                && e.seconds is int && e.seconds >= 1 && e.seconds <= 172800)
              || (e.type == 'oil'
                && e.keys().hasAll(['previousSeconds'])
                && e.byUid == request.auth.uid
                && e.byEmail == request.auth.token.email
                && e.previousSeconds is int && e.previousSeconds >= 0));
        }
        allow read: if pumpId in ['testo','value'] && isEmployee();
        allow create: if pumpId in ['testo','value'] && isImporter() && validEvent() && linkedToPump();
        allow update, delete: if false;
      }
    }
  }
}
